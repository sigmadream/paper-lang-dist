#include <iostream>

using namespace std;

int gcd(int a, int b) {
    while (b == 0) {
        int r = a % b;
        a = b;
        b = r;
    }
    return a;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int a, b;
    if (!(cin >> a >> b)) return 0;
    int g = gcd(a, b);
    cout << g << "\n" << (a / g) * b << "\n";
    return 0;
}
